const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", "Access-Control-Allow-Methods": "GET, OPTIONS", }; Deno.serve(async (req) => { if (req.method === "OPTIONS") { return new Response("ok", { headers: corsHeaders }); } try { const apiKey = Deno.env.get("LIGHTCAST_VOD_API_KEY"); if (!apiKey) { throw new Error("Lightcast API key is not configured."); } const channelId = "72119"; const lightcastUrl = new URL( "https://mediacloud.lightcast.com/api/channels" ); lightcastUrl.searchParams.set("action", "media"); lightcastUrl.searchParams.set("channelId", channelId); lightcastUrl.searchParams.set("apiKey", apiKey); const lightcastResponse = await fetch(lightcastUrl.toString(), { method: "GET", headers: { Accept: "application/json", }, }); const responseText = await lightcastResponse.text(); let lightcastData; try { lightcastData = JSON.parse(responseText); } catch { lightcastData = responseText; } if (!lightcastResponse.ok) { return new Response( JSON.stringify({ success: false, status: lightcastResponse.status, error: lightcastData, }), { status: 502, headers: { ...corsHeaders, "Content-Type": "application/json", }, } ); } return new Response( JSON.stringify({ success: true, channelId, data: lightcastData, }), { headers: { ...corsHeaders, "Content-Type": "application/json", "Cache-Control": "public, max-age=60", }, } ); } catch (error) { return new Response( JSON.stringify({ success: false, error: error instanceof Error ? error.message : "Unknown server error", }), { status: 500, headers: { ...corsHeaders, "Content-Type": "application/json", }, } ); } });
